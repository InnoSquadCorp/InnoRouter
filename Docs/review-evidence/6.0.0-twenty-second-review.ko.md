### 남은 배포 전 gate

- 실제 앱에 mount한 macOS modifier와 iOS Simulator root 재연결 runtime은 이번 실행에 포함하지 않았다. package production attach/detach 경로의 결정적 회귀는 통과했지만 UI mount 증거를 대신하지 않는다.
- pinned Xcode 26.6/Swift 6.3.3에서는 flag dialect fixture만 통과했다. 해당 toolchain 전체 suite·consumer·platform build는 미실행이다.
- exact commit 원격 CI, exact downstream revision, 실제 기기·접근성 검증은 commit이 없는 현재 후보에서 수행할 수 없거나 수행하지 않았다.
- `check-changelog-sync.sh`의 commit-range 검증도 같은 이유로 아직 유효한 PASS가 아니다. 현재 `CHANGELOG.md`의 Unreleased와 API baseline은 함께 수정했으며, commit 후 exact range로 다시 확인해야 한다.
- 따라서 제품 반례 수정과 필수 로컬 검증은 완료했지만, 이 문서만으로 release-ready·tag·발행 완료를 선언하지 않는다.

F01은 실제 modifier가 사용하는 package attach/detach 경로를 직접 호출한 증거다. 새로운 mounted UI 재현을 수행했다고 주장하지 않는다. F03은 5초 watchdog의 SIGALRM(exit 142)이 아니라 SIGSEGV(exit 139)이며, symbolicated stack trace는 확보하지 않았다. 종료 전 출력이 없어 crash runtime log는 비어 있었다.

기존 비재귀 `FeatureParentProbeRoute` 대조군은 부모/child에 동일한 `/leaf/:id`를 둬 의도된 ambiguity를 만든다. 그 nil을 "Feature는 원래 nil"의 근거로 삼지 않는다.

R6R22-Q01은 `RouterNineteenthReviewDeepLinkTests.swift`의 반환값 무시·count 자기 비교·실제로 중첩되지 않는 ternary와 `principle-gates.sh`의 종료 문자열 검사에서 확인했다. 저장 순서 테스트의 yield 기반 예약 추정, `try? await cancel.value`, 이전 증거의 PASS/미실행 충돌도 같은 검증 신뢰성 문제로 묶는다.

직전 검토에서 전체 `swift test --jobs 2 --no-parallel`는 575 tests/69 suites, local external consumer는 7 tests 및 positive/negative build, source lint/docs consistency는 통과했다. known issue 1건은 TestStore의 `withKnownIssue` 진단이다. 이 결과는 제품 반례 해결 증거가 아니다. Xcode 26.6/Swift 6.3.3·전 플랫폼 native runtime·exact SHA CI는 해당 검토에서 새로 확인하지 않았다.

import Foundation
import Synchronization
import InnoRouter
@testable import InnoRouterSwiftUI

enum ProbeRoute: String, Route, Codable { case saved, current }
final class Storage: RouterSnapshotStorage {
    let state: Mutex<(Data?, Int)>
    let loads: AsyncStream<Void>
    let signal: AsyncStream<Void>.Continuation
    let barrier = DispatchSemaphore(value: 0)
    init(_ data: Data) {
        state = Mutex((data, 0))
        (loads, signal) = AsyncStream.makeStream(of: Void.self)
    }
    func load() throws -> Data? {
        let count = state.withLock { $0.1 += 1; return $0.1 }
        signal.yield()
        if count == 1 {
            guard barrier.wait(timeout: .now() + 4) == .success else { throw CancellationError() }
        }
        return state.withLock { $0.0 }
    }
    func save(_ data: Data) throws { state.withLock { $0.0 = data } }
    func remove() throws { state.withLock { $0.0 = nil } }
}
@main struct Probe {
    @MainActor static func main() async throws {
        let codec = try RouterSnapshotCodec<ProbeRoute>(currentVersion: 1)
        let storage = Storage(try codec.encode(.rootStack(path: [.saved])))
        let store = RouterStore<ProbeRoute>()
        let driver = RouterRestorationDriver(store: store, codec: codec, storage: storage, saveDebounce: .seconds(3600))
        let firstOwner = UUID()
        let first = Task { @MainActor in try await driver.attach(firstOwner) }
        defer { first.cancel(); storage.barrier.signal(); driver.stop() }
        var iterator = storage.loads.makeAsyncIterator()
        _ = await iterator.next()
        if CommandLine.arguments.contains("control") {
            storage.barrier.signal()
            _ = try await first.value
        }
        driver.stop()
        storage.barrier.signal()
        _ = try? await first.value
        _ = await store.perform(.push(.current))
        let secondOwner = UUID()
        print("after.stop", try await driver.attach(secondOwner))
        print("before.detach", store.state, "revision", store.revision)
        driver.detach(secondOwner)
        let thirdOwner = UUID()
        print("after.reattach", try await driver.attach(thirdOwner))
        print("after.detach", store.state, "revision", store.revision)
        print("loadCount", storage.state.withLock { $0.1 })
    }
}

import Foundation

@Router(deepLinkSchemes: ["probe"], deepLinkHosts: ["app"])

    @FeatureRoute case child(Self)
    var destination: some View { EmptyView() }
}
@Router(deepLinkSchemes: ["probe"], deepLinkHosts: ["app"])
enum ChildRoute {
    @DeepLink("/child/:id") case child(id: String)
    var destination: some View { EmptyView() }
}
@Router(deepLinkSchemes: ["probe"], deepLinkHosts: ["app"])
enum ControlRoute {
    @DeepLink("/leaf/:id") case leaf(id: String)
    @FeatureRoute case feature(ChildRoute)
    var destination: some View { EmptyView() }
}
@main struct Probe {
    static func main() {
        let url = URL(string: "probe://app/leaf/42")!
        let origin = DeepLinkOrigin(scheme: "probe", host: "app")!

## 재실행 절차

아래 명령은 검토 당시 사용한 compiler/linker 구성을 기록한다. 먼저 본문의 각 코드 블록을 서로 다른 임시 Swift 파일로 복원하고 위 SHA-256을 확인한다. `PROBE_DIR`에는 복원한 세 파일이 있는 실제 임시 경로를 지정한다. 첫 `swift test`로 검사할 후보의 module/plugin을 빌드하고, 컴파일 실패를 runtime 결과와 혼동하지 않는다. F03의 비정상 종료는 수정 전 기대 결과이며 수정 후에는 정상 fail-closed 결과로 바뀌어야 한다.
